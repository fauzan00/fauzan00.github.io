// Entry function to manage input and output
function displayMostFrequent() {
  // 1. GET INPUT STRING
  const inputString = document.getElementById("numberInput").value;

  let numsArray = [];
  let arrayIndex = 0;
  let currentNumberString = "";

  // Helper function to process the string built so far
  function processAndStoreNumber(str) {
    let processedStr = "";
    let k = 0;

    // Manual trim (remove spaces)
    while (str[k] !== undefined) {
      const char = str[k];
      if (char !== " ") {
        processedStr = processedStr + char;
      }
      k++;
    }

    // Validate: check if all characters are digits or minus sign
    if (processedStr !== "") {
      let isValid = true;
      let m = 0;
      while (processedStr[m] !== undefined) {
        const c = processedStr[m];
        if (m === 0 && c === "-") {
          // Allow minus at start
        } else if (c < "0" || c > "9") {
          isValid = false;
          break;
        }
        m++;
      }

      if (isValid && processedStr !== "-") {
        const num = parseInt(processedStr);
        if (!isNaN(num)) {
          numsArray[arrayIndex] = num;
          arrayIndex++;
        }
      }
    }
  }

  // 2. CONVERT STRING TO ARRAY OF NUMBERS (Manual Parsing)
  let i = 0;
  while (inputString[i] !== undefined) {
    const char = inputString[i];

    if (char === ",") {
      processAndStoreNumber(currentNumberString);
      currentNumberString = "";
    } else {
      currentNumberString = currentNumberString + char;
    }
    i++;
  }

  // Process the last number
  processAndStoreNumber(currentNumberString);

  // Check if array is empty
  if (numsArray[0] === undefined) {
    document.getElementById("demo").innerHTML =
      "Invalid or empty input. Please enter comma-separated numbers.";
    return;
  }

  // 3. CALL CORE LOGIC
  const result = findMostFrequent(numsArray);

  // 4. DISPLAY RESULT
  document.getElementById("demo").innerHTML = 
    `The most frequent number is: ${result}`;
}

// Core logic function
function findMostFrequent(nums) {
  let numbers = [];
  let counts = [];
  let size = 0;

  // Loop 1: Count occurrences of each number
  let i = 0;
  while (nums[i] !== undefined) {
    const currentNum = nums[i];
    let found = false;
    let index = -1;

    // Check if currentNum exists in numbers array
    let j = 0;
    while (j < size) {
      if (numbers[j] === currentNum) {
        found = true;
        index = j;
        break;
      }
      j++;
    }

    if (!found) {
      numbers[size] = currentNum;
      counts[size] = 1;
      size++;
    } else {
      counts[index]++;
    }
    i++;
  }

  // Loop 2: Find the maximum count
  let maxCount = -1;
  let mostFrequentNumber = null;

  let k = 0;
  while (k < size) {
    if (counts[k] > maxCount) {
      maxCount = counts[k];
      mostFrequentNumber = numbers[k];
    }
    k++;
  }

  return mostFrequentNumber;
}
