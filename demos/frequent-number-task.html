Task- Get the Frequent number

Given a random non-empty array of integers, write a script that is able to output the integer number that appears the most.

Kindly do not use any library functions to directly return the most common number, as doing so will result in immediate failure.

Example:
Input:nums =[3,2,2,2,2,2,3,4,5] Output:2

Input:nums =[1,2,3,4,5,5] Output:5

Answer and Explanation

Algorithm Approach (No Library Functions):
This solution uses only basic loops and comparisons - no indexOf(), map(), filter(), or other array methods.

Step 0: Parse String into Array (Manual Chunking)
Separate comma-separated string into array of numbers:
• Iterate through string character by character
• Build current number string until comma is found
• When comma found → trim spaces, convert to number, add to array
• Process last number after loop ends
• Result: Array of numbers ready for processing

Step 1: Build a List by Counting Occurrences
Iterate through input array and build two parallel lists:
• numbers[] - stores unique numbers found
• counts[] - stores how many times each number appears
• For each number: check if it exists in list (iterate through numbers[])
  - If new → add to both lists, increment size
  - If exists → increment its count

Step 2: Find Maximum by Listing Through Counts
Iterate through the counts list to find the highest value:
• Initialize: maxCount = -1, mostFrequentNumber = null
• List through each count: if counts[k] > maxCount
  - Update maxCount and mostFrequentNumber

Result: mostFrequentNumber is the answer

JavaScript Implementation: 
// STEP 0: Parse comma-separated string into array
function parseNumbers(inputString) {
  let numsArray = [];
  let arrayIndex = 0;
  let currentNumberString = "";
  
  let i = 0;
  while (inputString[i] !== undefined) {  // Iterate char by char
    const char = inputString[i];
    
    if (char === ',') {  // Found comma separator
      // Trim spaces manually
      let processedStr = "";
      let k = 0;
      while (currentNumberString[k] !== undefined) {
        if (currentNumberString[k] !== ' ') {
          processedStr += currentNumberString[k];
        }
        k++;
      }
      
      // Convert to number and add to array
      if (processedStr !== "") {
        const num = parseInt(processedStr);
        if (!isNaN(num)) {
          numsArray[arrayIndex] = num;
          arrayIndex++;
        }
      }
      currentNumberString = "";  // Reset for next number
    } else {
      currentNumberString += char;  // Build number string
    }
    i++;
  }
  
  // Process last number (no comma after it)
  if (currentNumberString !== "") {
    let processedStr = "";
    let k = 0;
    while (currentNumberString[k] !== undefined) {
      if (currentNumberString[k] !== ' ') {
        processedStr += currentNumberString[k];
      }
      k++;
    }
    if (processedStr !== "") {
      const num = parseInt(processedStr);
      if (!isNaN(num)) {
        numsArray[arrayIndex] = num;
      }
    }
  }
  
  return numsArray;
}

// MAIN ALGORITHM: Find most frequent number
function findMostFrequent(nums) {
  // Initialize two parallel arrays and size counter
  let numbers = [];  // Stores unique numbers
  let counts = [];   // Stores count for each number
  let size = 0;      // Manual size tracking
  
  // LOOP 1: Build list by counting occurrences
  let i = 0;
  while (nums[i] !== undefined) {  // Iterate through input
    const currentNum = nums[i];
    let found = false;
    let index = -1;
    
    // Inner loop: Check if currentNum exists in numbers[]
    let j = 0;
    while (j < size) {  // List through existing numbers
      if (numbers[j] === currentNum) {
        found = true;   // Number found
        index = j;      // Store its position
        break;          // Exit inner loop
      }
      j++;  // Move to next position
    }
    
    // Process based on whether number was found
    if (!found) {
      // New number: add to both lists
      numbers[size] = currentNum;
      counts[size] = 1;
      size++;  // Increment size counter
    } else {
      // Existing number: increment its count
      counts[index]++;
    }
    i++;  // Move to next input number
  }
  
  // LOOP 2: List through counts to find maximum
  let maxCount = -1;              // Track highest count
  let mostFrequentNumber = null;  // Track number with highest count
  
  let k = 0;
  while (k < size) {  // Iterate through all counts
    if (counts[k] > maxCount) {
      // Found new maximum
      maxCount = counts[k];
      mostFrequentNumber = numbers[k];
    }
    k++;  // Move to next count
  }
  
  return mostFrequentNumber;  // Return the result
}
